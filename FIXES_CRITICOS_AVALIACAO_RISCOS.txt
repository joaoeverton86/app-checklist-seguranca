```javascript
async function salvarRiscoGheEmEdicao(idx) {
    const statusEl = document.getElementById('matrizRiscoSaveStatus');
    const g = allGheCatalogo.find(x => String(x.id) === String(matrizGheSelecionadoId));
    if (!g) return;

    // VALIDAÇÃO DE CAMPOS OBRIGATÓRIOS (usa cópia de trabalho se existir)
    const r = (matrizRiscoEditIdx === idx && matrizRiscoEditForm.length > 0) ? matrizRiscoEditForm[idx] : g.riscos[idx];
    const erros = [];
    if (!r.agente?.trim()) erros.push('Nome do Agente/Fator de Risco');
    if (!r.tipo_agente?.trim()) erros.push('Tipo de Agente');
    if (!r.probabilidade) erros.push('Probabilidade (1-5)');
    if (!r.severidade) erros.push('Severidade (1-5)');
    
    if (erros.length > 0) {
        if (statusEl) { 
            statusEl.textContent = '❌ Preencha: ' + erros.join(', '); 
            statusEl.style.color = 'var(--danger)'; 
        }
        return;
    }

    if (statusEl) { statusEl.textContent = 'Salvando...'; statusEl.style.color = 'var(--text-light)'; }
    try {
        // Aplica cópia de trabalho no objeto principal antes de salvar
        if (matrizRiscoEditIdx === idx && matrizRiscoEditForm.length > 0) {
            g.riscos = matrizRiscoEditForm;
            matrizRiscoEditForm = [];
            matrizRiscoEditIdx = -1;
        }
        
        const payload = [{
            id: g.id,
            nome: g.nome,
            cargos: g.cargos || [],
            quantidade_oficial: g.quantidade_oficial || 0,
            riscos: g.riscos || [],
            conclusoes: g.conclusoes || {},
            updated_at: new Date().toISOString()
        }];

        await supabaseUpsert('ghe_catalogo', payload);
        if (statusEl) { statusEl.textContent = '✅ Salvo com sucesso!'; statusEl.style.color = 'var(--success)'; }
        renderMatrizViewSetor();
    } catch (err) {
        console.error('Erro ao salvar risco do GHE:', err);
        if (statusEl) { statusEl.textContent = '❌ Falha: ' + err.message; statusEl.style.color = 'var(--danger)'; }
    }
}
```

```javascript
// ============================================
var matrizRiscoEditForm = [];
var matrizRiscoEditIdx = -1;

function cancelarEdicaoRiscoGhe(idx) {
    const g = allGheCatalogo.find(x => String(x.id) === String(matrizGheSelecionadoId));
    if (!g || !Array.isArray(g.riscos) || !g.riscos[idx]) return;
    
    // Recarrega dados originais do Supabase para descartar alterações locais
    supabaseFetch('ghe_catalogo', '?id=eq.' + encodeURIComponent(g.id) + '&select=riscos')
        .then(rows => {
            if (rows && rows.length > 0 && Array.isArray(rows[0].riscos)) {
                g.riscos = rows[0].riscos;
                matrizRiscoEditForm = [];
                matrizRiscoEditIdx = -1;
                renderMatrizViewSetor();
            }
        })
        .catch(err => {
            console.error('Erro ao recarregar riscos originais:', err);
            // Fallback: re-renderiza com dados atuais em memória
            matrizRiscoEditForm = [];
            matrizRiscoEditIdx = -1;
            renderMatrizViewSetor();
        });
}

// Substitui atualizarCampoRiscoEmEdicao para usar cópia de trabalho:
function atualizarCampoRiscoEmEdicao(idx, campo, valor) {
    const g = allGheCatalogo.find(x => String(x.id) === String(matrizGheSelecionadoId));
    if (!g || !Array.isArray(g.riscos) || !g.riscos[idx]) return;
    
    // Inicializa cópia de trabalho se não existe ou mudou de risco
    if (matrizRiscoEditIdx !== idx || matrizRiscoEditForm.length === 0) {
        matrizRiscoEditIdx = idx;
        matrizRiscoEditForm = g.riscos.map(r => ({ ...r }));
    }
    
    matrizRiscoEditForm[idx][campo] = valor;
    
    // Se for probabilidade ou severidade, recalcula gravidade na cópia + atualiza badge visual
    if (campo === 'probabilidade' || campo === 'severidade') {
        atualizarGravidadeCalculadaGhe(matrizRiscoEditForm[idx]);
        // Atualiza badge de gravidade no painel de detalhes sem re-renderizar tudo
        const badgeEl = document.querySelector('#matrizRiscoDetalhesContainer [style*="border-radius:8px"][style*="font-weight:800"]');
        if (badgeEl) {
            const nivel = nivelRiscoGhe(matrizRiscoEditForm[idx].probabilidade, matrizRiscoEditForm[idx].severidade);
            const corGrav = nivel.cor || '#888888';
            badgeEl.innerHTML = `Gravidade ${nivel.valor !== null ? nivel.valor : (matrizRiscoEditForm[idx].gravidade || '—')} — ${escapeHTML(nivel.label || matrizRiscoEditForm[idx].gravidade_label || 'Pendente')}`;
            badgeEl.style.background = corGrav + '22';
            badgeEl.style.borderColor = corGrav;
            badgeEl.style.color = corGrav;
        }
    }
    
    // Se mudou tipo_agente, precisa atualizar o select de agentes (catálogo)
    if (campo === 'tipo_agente') {
        const selectAgente = document.getElementById('riscoEdit_agente_select');
        const inputAgente = document.getElementById('riscoEdit_agente');
        const inputEsocial = document.getElementById('riscoEdit_esocial');
        if (selectAgente) {
            // Reconstroi opções do catálogo para o novo tipo
            const opcoes = getEsocialOpcoesPorTipo(valor);
            selectAgente.innerHTML = '<option value="">-- Selecione no catálogo ou digite abaixo --</option>' +
                opcoes.map(a => `<option value="${escapeHTML(a.nome)}" data-codigo="${escapeHTML(a.codigo)}">${escapeHTML(a.nome)}</option>`).join('') +
                '<option value="__CUSTOM__">➕ Outro (digitar manualmente)...</option>';
            // Limpa agente e eSocial selecionados
            if (inputAgente) { inputAgente.value = ''; inputAgente.style.display = 'none'; }
            if (inputEsocial) { inputEsocial.value = ''; }
            atualizarCampoRiscoEmEdicao(idx, 'agente', '');
            atualizarCampoRiscoEmEdicao(idx, 'esocial_codigo', '');
        }
    }
}
```

```javascript
function toggleEpiSelecaoBusca(checkbox, epi) {
    const idx = ${idx}; // This will be replaced by the actual idx value
    if (!window.epiSelecionadosBusca_${idx}) window.epiSelecionadosBusca_${idx} = [];
    const arr = window.epiSelecionadosBusca_${idx};
    const jaTem = arr.find(e => e.ca === epi.ca);
    if (checkbox.checked && !jaTem) arr.push(epi);
    else if (!checkbox.checked) {
        const i = arr.findIndex(e => e.ca === epi.ca);
        if (i >= 0) arr.splice(i, 1);
    }
    const countEl = document.getElementById('epiSelecionadosCount_${idx}');
```
